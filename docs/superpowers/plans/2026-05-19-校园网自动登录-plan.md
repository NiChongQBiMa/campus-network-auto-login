```python
import configparser
import os
import sys

def get_config_path():
    """获取 config.ini 路径，与 exe 同目录"""
    if getattr(sys, 'frozen', False):
        return os.path.join(os.path.dirname(sys.executable), 'config.ini')
    else:
        return os.path.join(os.path.dirname(os.path.abspath(__file__)), 'config.ini')

def load_config():
    """读取配置文件"""
    config = configparser.ConfigParser()
    config_path = get_config_path()
    if not os.path.exists(config_path):
        raise FileNotFoundError(f"配置文件不存在：{config_path}")
    config.read(config_path, encoding='utf-8')
    return config

def save_form_config(fields):
    """首次分析表单后，将字段映射写入 [form] 段"""
    config = load_config()
    if 'form' not in config:
        config.add_section('form')
    for key, value in fields.items():
        config.set('form', key, value)
    with open(get_config_path(), 'w', encoding='utf-8') as f:
        config.write(f)
```

import socket
import requests

def get_local_ip():
    """检查本机是否有有效IP（排除APIPA自动配置地址）"""
    try:
        hostname = socket.gethostname()
        ip = socket.gethostbyname(hostname)
        if ip.startswith('127.') or ip.startswith('169.254.'):
            return None
        return ip
    except:
        return None

def tcp_check(host, port=80, timeout=3):
    """TCP连接检测，3秒超时"""
    try:
        sock = socket.create_connection((host, port), timeout=timeout)
        sock.close()
        return True
    except:
        return False

def http_check(url, timeout=5):
    """HTTP GET确认页面可达"""
    try:
        r = requests.get(url, timeout=timeout, allow_redirects=True)
        return r.status_code == 200
    except:
        return False

def wait_for_network(host, port, timeout, interval):
    """循环等待网络就绪，返回True=就绪 False=超时"""
    elapsed = 0
    while elapsed < timeout:
        if get_local_ip() and tcp_check(host, port):
            return True
        time.sleep(interval)
        elapsed += interval
    return False
```

from bs4 import BeautifulSoup
import re

def analyze_login_form(html, username, password, save_password):
    """解析登录页面HTML，返回 (action_url, post_data)"""
    soup = BeautifulSoup(html, 'html.parser')
    forms = soup.find_all('form')
    if not forms:
        raise Exception("页面中未找到登录表单")
    
    form = forms[0]
    action = form.get('action', '')
    base_url = get_base_url_from_config()
    
    # 拼接完整 action URL
    if action.startswith('http'):
        action_url = action
    elif action.startswith('/'):
        from urllib.parse import urlparse
        parsed = urlparse(base_url)
        action_url = f"{parsed.scheme}://{parsed.netloc}{action}"
    else:
        action_url = base_url.rstrip('/') + '/' + action.lstrip('/')
    
    # 收集所有 input 字段
    form_fields = {}
    post_data = {}
    
    for inp in form.find_all('input'):
        name = inp.get('name', '')
        input_type = inp.get('type', 'text').lower()
        value = inp.get('value', '')
        
        if input_type == 'hidden':
            post_data[name] = value
            continue
        
        if input_type == 'password':
            form_fields['password_field'] = name
            post_data[name] = password
        elif input_type == 'text' and not form_fields.get('username_field'):
            # 智能匹配用户名字段
            if re.search(r'user|account|name|id|uname|uid', name, re.I):
                form_fields['username_field'] = name
                post_data[name] = username
        elif input_type == 'checkbox':
            if re.search(r'save|remember|keep', name, re.I):
                form_fields['save_field'] = name
                if save_password:
                    post_data[name] = '1'
        
        if input_type == 'submit':
            form_fields['submit_field'] = name
    
    # 如果没匹配到用户名字段，取第一个 text 类型
    if 'username_field' not in form_fields:
        for inp in form.find_all('input'):
            if inp.get('type', 'text').lower() == 'text' and inp.get('name'):
                form_fields['username_field'] = inp['name']
                post_data[inp['name']] = username
                break
    
    form_fields['action_url'] = action_url
    # 返回字段映射供写入 config
    return form_fields, post_data, action_url
```

```python
def quick_login(config):
    """使用已缓存的表单字段直接登录"""
    login_url = config.get('server', 'login_url')
    username = config.get('account', 'username')
    password = config.get('account', 'password')
    
    action_url = config.get('form', 'action_url')
    
    post_data = {
        config.get('form', 'username_field'): username,
        config.get('form', 'password_field'): password,
    }
    if config.get('account', 'save_password') == '1' and config.has_option('form', 'save_field'):
        post_data[config.get('form', 'save_field')] = '1'
    
    # 先 GET 页面获取可能的 CSRF token
    session = requests.Session()
    resp = session.get(login_url, timeout=10)
    soup = BeautifulSoup(resp.text, 'html.parser')
    
    # 收集隐藏字段
    for inp in soup.find_all('input', type='hidden'):
        name = inp.get('name')
        value = inp.get('value', '')
        if name and name not in post_data:
            post_data[name] = value
    
    # POST 登录
    resp = session.post(action_url, data=post_data, timeout=10)
    return check_login_success(resp.text)

def check_login_success(html):
    """检查登录是否成功"""
    success_keywords = ['成功', 'success', '已登录', '在线', 'login success', '认证成功']
    for kw in success_keywords:
        if kw.lower() in html.lower():
            return True
    return False
```

import tkinter as tk
from tkinter import messagebox
import threading
from datetime import datetime

def show_popup(title, message, auto_close=0):
    """显示弹窗，auto_close>0时N秒后自动关闭"""
    def _show():
        root = tk.Tk()
        root.withdraw()  # 隐藏主窗口
        root.attributes('-topmost', True)
        if auto_close > 0:
            root.after(auto_close * 1000, root.destroy)
        messagebox.showinfo(title, message)
        root.destroy()
    
    # 在独立线程中运行 tkinter
    t = threading.Thread(target=_show, daemon=True)
    t.start()
    t.join(timeout=auto_close + 5 if auto_close else None)
```

import logging

def setup_logging():
    """配置日志"""
    log_path = os.path.join(os.path.dirname(get_config_path()), 'login.log')
    logging.basicConfig(
        level=logging.INFO,
        format='%(asctime)s [%(levelname)s] %(message)s',
        datefmt='%Y-%m-%d %H:%M:%S',
        handlers=[
            logging.FileHandler(log_path, encoding='utf-8'),
        ]
    )

def main():
    setup_logging()
    logging.info("校园网自动登录程序启动")
    
    try:
        config = load_config()
    except FileNotFoundError as e:
        show_popup("错误", f"配置文件缺失\n{e}")
        logging.error(str(e))
        sys.exit(1)
    
    # 读取配置
    login_url = config.get('server', 'login_url')
    host = login_url.split('://')[-1].split(':')[0].split('/')[0]
    port = 80
    if ':' in login_url.split('://')[-1].split('/')[0]:
        port = int(login_url.split('://')[-1].split('/')[0].split(':')[1])
    
    timeout = config.getint('settings', 'wait_network_timeout')
    interval = config.getint('settings', 'retry_interval')
    
    # 等待网络
    logging.info(f"等待网络就绪（超时{timeout}秒）...")
    if not wait_for_network(host, port, timeout, interval):
        msg = f"网络连接超时，{host} 不可达"
        show_popup("网络超时 ⏰", msg)
        logging.error(msg)
        sys.exit(1)
    
    logging.info("网络已就绪")
    
    # 检查是否有缓存的表单配置
    form_configured = config.has_option('form', 'action_url') and config.get('form', 'action_url')
    
    if not form_configured:
        # 首次运行：分析表单
        logging.info("首次运行，分析登录表单...")
        try:
            resp = requests.get(login_url, timeout=10)
            username = config.get('account', 'username')
            password = config.get('account', 'password')
            save_pwd = config.get('account', 'save_password') == '1'
            
            fields, post_data, action_url = analyze_login_form(resp.text, username, password, save_pwd)
            save_form_config(fields)
            
            # 用分析出的数据登录
            session = requests.Session()
            session.post(action_url, data=post_data, timeout=10)
            # 再GET一次确认
            check_resp = session.get(login_url, timeout=10)
            success = check_login_success(check_resp.text)
        except Exception as e:
            logging.error(f"表单分析失败: {e}")
            show_popup("登录失败 ❌", f"无法分析登录表单\n{e}")
            sys.exit(1)
    else:
        # 使用缓存配置登录
        logging.info("使用缓存配置登录...")
        try:
            success = quick_login(config)
        except Exception as e:
            logging.error(f"登录失败: {e}")
            show_popup("登录失败 ❌", f"登录请求失败\n{e}")
            sys.exit(1)
    
    if success:
        now = datetime.now().strftime('%Y-%m-%d %H:%M:%S')
        msg = f"校园网已连接 ✅\n当前时间：{now}"
        show_popup("登录成功", msg, auto_close=10)
        logging.info("登录成功")
    else:
        show_popup("登录失败 ❌", "请检查账号密码是否正确\n详情见 login.log")
        logging.error("登录失败：返回页面未检测到成功标志")

if __name__ == '__main__':
    main()
```
